Koneksi Google Spreadsheet

Tempelkan URL Web App dari Google Apps Script di bawah ini agar data siswa otomatis tersimpan & terbaca dari Google Spreadsheet Anda.

Langkah Mudah Menghubungkan ke Google Spreadsheet:

  1. Buka sheets.new untuk membuat Spreadsheet baru.
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Hapus kode yang ada, lalu tempel kode di bawah ini dan klik Simpan (Save).
  4. Klik Terapkan (Deploy) > Deployment baru (New deployment) > Pilih jenis Aplikasi Web (Web app).
  5. Pada "Yang memiliki akses (Who has access)", pilih Siapa saja (Anyone), lalu klik Terapkan & salin URL Web App-nya ke kotak di atas.
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var action = e.parameter.action;
  if (action == "delete") {
    var nameToDelete = e.parameter.name;
    var data = sheet.getDataRange().getValues();
    for (var i = data.length - 1; i >= 1; i--) {
      if (data[i][0].toString().toLowerCase() == nameToDelete.toLowerCase()) {
        sheet.deleteRow(i + 1);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "deleted"})).setMimeType(ContentService.MimeType.JSON);
  }
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    result.push({ name: rows[i][0], gender: rows[i][1], score: Number(rows[i][2]), timeSeconds: Number(rows[i][3]), date: rows[i][4] });
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Nama", "Jenis Kelamin", "Skor", "Waktu (Detik)", "Tanggal"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.name, data.gender, data.score, data.timeSeconds, data.date]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
